file uploding at local and live in react
import streamlit as st import pandas as pd import sqlite3 from datetime import datetime import urllib.parse  # --- 1. إعداد قاعدة البيانات --- conn = sqlite3.connect('business_pro_v4.db') c = conn.cursor()  # إنشاء الجداول (مخزن، عملاء، فواتير) c.execute('CREATE TABLE IF NOT EXISTS inventory (item TEXT PRIMARY KEY, price REAL, qty INTEGER, category TEXT, color TEXT)') c.execute('CREATE TABLE IF NOT EXISTS customers (phone TEXT PRIMARY KEY, name TEXT, address TEXT, debt REAL)') c.execute('CREATE TABLE IF NOT EXISTS invoices (id TEXT, cust_name TEXT, total REAL, paid REAL, type TEXT, date TEXT)') conn.commit()  # --- 2. وظائف مساعدة --- if 'cart' not in st.session_state:     st.session_state.cart = {}  def add_to_cart(item, price):     if item in st.session_state.cart:         st.session_state.cart[item]['qty'] += 1     else:         st.session_state.cart[item] = {'price': price, 'qty': 1}  # --- 3. تصميم واجهة التطبيق --- st.set_page_config(page_title="نظام الإدارة الذكي", layout="wide")  # تنبيه للمخزن المنخفض (أقل من 5 قطع) low_stock = c.execute("SELECT item, qty FROM inventory WHERE qty < 5").fetchall() if low_stock:     for item in low_stock:         st.error(f"⚠️ تنبيه: الصنف {item[0]} أوشك على النفاد (المتبقي: {item[1]})")  # --- القسم الأول: بيانات العميل والشحن --- st.title("🛍️ نقطة البيع الاحترافية") with st.expander("👤 بيانات العميل والتوصيل (اضغط للفتح)", expanded=True):     col_c1, col_c2, col_c3 = st.columns(3)     with col_c1:         c_name = st.text_input("اسم العميل")         c_phone = st.text_input("رقم الهاتف (واتساب)")     with col_c2:         c_address = st.text_input("العنوان بالتفصيل")         driver = st.text_input("اسم السائق")     with col_c3:         delivery_fee = st.number_input("رسوم التوصيل", min_value=0.0)         inv_type = st.selectbox("نوع الفاتورة", ["مبيعات", "مرتجعات", "فاتورة مؤقتة"])  # --- القسم الثاني: نظام المجلدات والأزرار الملونة --- st.subheader("📦 اختر البنود (نقرة واحدة للإضافة)") categories = ["عام", "ألبان", "منظفات", "مشروبات"] tabs = st.tabs(categories)  for i, cat in enumerate(categories):     with tabs[i]:         # جلب المنتجات من القاعدة         items = c.execute("SELECT item, price, color, qty FROM inventory WHERE category=?", (cat,)).fetchall()         if not items:             st.info("لا توجد أصناف في هذا المجلد حالياً.")         else:             cols = st.columns(4)             for idx, item in enumerate(items):                 btn_label = f"{item[0]}\n{item[1]} ريال"                 if cols[idx % 4].button(btn_label, key=f"btn_{item[0]}", use_container_width=True):                     add_to_cart(item[0], item[1])  # --- القسم الثالث: الفاتورة والحسابات --- st.divider() st.subheader("🧾 مراجعة الفاتورة")  if st.session_state.cart:     total_items_price = 0     for item_name, info in list(st.session_state.cart.items()):         col_f1, col_f2, col_f3, col_f4 = st.columns([3, 2, 1, 1])         with col_f1:             st.write(f"**{item_name}**")         with col_f2:             # أزرار + و -             c_m, c_v, c_p = st.columns(3)             if c_m.button("➖", key=f"min_{item_name}"):                 st.session_state.cart[item_name]['qty'] -= 1                 if st.session_state.cart[item_name]['qty'] <= 0: del st.session_state.cart[item_name]                 st.rerun()             c_v.write(info['qty'])             if c_p.button("➕", key=f"pls_{item_name}"):                 st.session_state.cart[item_name]['qty'] += 1                 st.rerun()                  item_total = info['price'] * info['qty']         total_items_price += item_total         with col_f3:             st.write(f"{item_total} ريال")         with col_f4:             if st.button("🗑️", key=f"del_{item_name}"):                 del st.session_state.cart[item_name]                 st.rerun()      # الحسابات النهائية     st.divider()     col_res1, col_res2 = st.columns(2)     with col_res1:         discount_per = st.number_input("نسبة الخصم (%)", min_value=0, max_value=100, value=0)         subtotal = total_items_price + delivery_fee         final_total = subtotal * (1 - discount_per/100)         st.markdown(f"### الإجمالي النهائي: {final_total:.2f} ريال")          with col_res2:         if c_name:             paid = st.number_input("المبلغ المدفوع", min_value=0.0)             remaining = final_total - paid             if remaining > 0:                 st.warning(f"المتبقي (آجل): {remaining:.2f}")             else:                 st.success("الفاتورة مدفوعة بالكامل")      # زر الحفظ والإرسال     if st.button("✅ اعتماد الفاتورة وحفظ البيانات"):         inv_id = datetime.now().strftime("%Y%m%d%H%M")         # حفظ الفاتورة في القاعدة         c.execute("INSERT INTO invoices VALUES (?,?,?,?,?,?)", (inv_id, c_name, final_total, paid, inv_type, datetime.now().date()))         # تحديث مديونية العميل         if c_name and remaining > 0:             c.execute("INSERT OR REPLACE INTO customers VALUES (?, ?, ?, IFNULL((SELECT debt FROM customers WHERE phone=?), 0) + ?)",                        (c_phone, c_name, c_address, c_phone, remaining))         conn.commit()                  st.success("تم حفظ البيانات في القاعدة بنجاح!")                  # رابط الواتساب         msg = f"فاتورة {inv_type}\nالعميل: {c_name}\nالإجمالي: {final_total}\nالمدفوع: {paid}\nالمتبقي: {remaining}"         whatsapp_url = f"https://api.whatsapp.com/send?phone={c_phone}&text={urllib.parse.quote(msg)}"         st.markdown(f'[📲 اضغط هنا لإرسال الفاتورة عبر الواتساب]({whatsapp_url})', unsafe_allow_html=True)  else:     st.info("السلة فارغة، انقر على المنتجات في الأعلى للبدء.")  # --- القسم الرابع: ملخص الديون --- st.divider() if st.checkbox("🔍 إظهار سجل الديون (من عليه فلوس)"):     debts_df = pd.read_sql_query("SELECT name as 'الاسم', phone as 'الهاتف', debt as 'الدين متبقي' FROM customers WHERE debt > 0", conn)     st.table(debts_df)
